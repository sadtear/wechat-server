公开区：接入流程与 API

这里不需要密钥，适合直接发给第三方开发者阅读。

极简接入流程

1
网站展示公众号入口登录页展示公众号二维码/名称，并提示用户发送配置的触发词。
2
用户获取验证码用户关注公众号后发送触发词，例如“验证码 / 登录 / code”。
3
后端换取 OpenID你的网站后端调用 /api/wechat/user，拿到 OpenID 后登录、注册或绑定站内用户。

接口说明

验证用户
GET /api/wechat/user?code=用户验证码&app_id=wx66ebe06eb1729498
Header: Authorization: Bearer {api_token}，成功后 data 为用户 OpenID。
服务状态
GET /api/wechat/stats
查看活跃验证码、活跃用户、已配置公众号列表。
配置管理
GET/POST /api/config
需要密钥，仅管理员在高级工具箱中使用。

接入代码示例

代码都收纳在 Tabs 里，复制对应语言即可。生产环境请把 API 密钥放在后端，不要暴露给前端。

// npm i express
import express from 'express';

const app = express();
const WECHAT_SERVER = process.env.WECHAT_SERVER || 'https://你的-wechat-server-域名';
const API_TOKEN = process.env.WECHAT_API_TOKEN || 'sakura';
const APP_ID = 'wx66ebe06eb1729498';

const usersByOpenId = new Map();
let nextUserId = 1;

app.get('/auth/wechat/callback', async (req, res) => {
  const code = String(req.query.code || '').trim();
  if (!code) return res.status(400).json({ error: '验证码不能为空' });

  const url = `${WECHAT_SERVER}/api/wechat/user?code=${encodeURIComponent(code)}&app_id=${APP_ID}`;
  const resp = await fetch(url, { headers: { Authorization: `Bearer ${API_TOKEN}` } });
  const json = await resp.json();
  if (!json.success) return res.status(401).json({ error: json.message });

  const openId = json.data;
  let user = usersByOpenId.get(openId);
  if (!user) {
    user = { id: nextUserId++, openId, nickname: `微信用户${nextUserId}` };
    usersByOpenId.set(openId, user);
  }
  res.json({ success: true, user });
});

app.listen(8080);

import os
import requests
from flask import Flask, request, jsonify

app = Flask(__name__)
WECHAT_SERVER = os.getenv('WECHAT_SERVER', 'https://你的-wechat-server-域名')
API_TOKEN = os.getenv('WECHAT_API_TOKEN', 'sakura')
APP_ID = 'wx66ebe06eb1729498'
users_by_openid = {}

@app.get('/auth/wechat/callback')
def wechat_callback():
    code = request.args.get('code', '').strip()
    if not code:
        return jsonify(error='验证码不能为空'), 400

    resp = requests.get(
        f'{WECHAT_SERVER}/api/wechat/user',
        params={'code': code, 'app_id': APP_ID},
        headers={'Authorization': f'Bearer {API_TOKEN}'},
        timeout=10,
    )
    data = resp.json()
    if not data.get('success'):
        return jsonify(error=data.get('message', '验证失败')), 401

    openid = data['data']
    user = users_by_openid.setdefault(openid, {'openid': openid, 'name': '微信用户'})
    return jsonify(success=True, user=user)
package main

import (
  "encoding/json"
  "net/http"
  "net/url"
  "os"
  "strings"
)

type WechatResp struct {
  Success bool   `json:"success"`
  Message string `json:"message"`
  Data    string `json:"data"`
}

func WechatLogin(w http.ResponseWriter, r *http.Request) {
  wechatServer := os.Getenv("WECHAT_SERVER")
  if wechatServer == "" { wechatServer = "https://你的-wechat-server-域名" }
  api := strings.TrimRight(wechatServer, "/") + "/api/wechat/user?" + url.Values{
    "code":   {r.URL.Query().Get("code")},
    "app_id": {"wx66ebe06eb1729498"},
  }.Encode()

  req, _ := http.NewRequest(http.MethodGet, api, nil)
  req.Header.Set("Authorization", "Bearer sakura")
  resp, err := http.DefaultClient.Do(req)
  if err != nil { http.Error(w, err.Error(), 500); return }
  defer resp.Body.Close()

  var out WechatResp
  json.NewDecoder(resp.Body).Decode(&out)
  if !out.Success { http.Error(w, out.Message, 401); return }
  json.NewEncoder(w).Encode(map[string]any{"success": true, "openid": out.Data})
}
CREATE TABLE users (
  id BIGINT PRIMARY KEY AUTO_INCREMENT,
  username VARCHAR(64),
  wechat_openid VARCHAR(128) UNIQUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
